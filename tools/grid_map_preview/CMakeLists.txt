cmake_minimum_required(VERSION 3.16)
project(nav_grid_map_preview LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(grid_map_core REQUIRED)
find_package(grid_map_pcl REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
add_executable(nav_grid_map_pcl_export export_elevation.cpp)
target_compile_features(nav_grid_map_pcl_export PRIVATE cxx_std_17)
target_include_directories(nav_grid_map_pcl_export PRIVATE ${PCL_INCLUDE_DIRS})
ament_target_dependencies(nav_grid_map_pcl_export rclcpp grid_map_core grid_map_pcl)
target_link_libraries(nav_grid_map_pcl_export ${PCL_LIBRARIES})
